Types of Services:

ClusterIP( default service type) : It exposes the Service on a stable, internal cluster IP address, making it only reachable from within the cluster. 

NodePort: exposes the Service on a static port. on each Node's IP address in the cluster

LoadBalancer:exposes the Service externally using a cloud provider's load balancer

ExternalName : special type that maps a Service to an external DNS name

Pipeline Script : created using Groovy language

pipeline {
    agent any // Specifies where the pipeline will run
    tools{
	jdk 'Java17'
        maven 'Maven'
    }
    stages {
        stage('Checkout the code from github') {
            steps {
                echo 'Checkout the code from github...'
                git branch: 'main', credentialsId: 'gitcred', url: 'https://github.com/chntraining/k8bnp7.git'
            }
        }
        stage('Build the application') {
            steps {
                echo 'Building the application...' 
                bat 'mvn clean package'
            }
        }
        stage('Build the Docker Image') {
            steps {
                echo 'Building Docker Image ....'
                bat 'docker build -t pms-app:latest .'
            }
        }
        
        stage('Start the minikube') {
            steps {
		echo 'Deleting Minikube'
		bat 'minikube delete'

                echo 'Starting Minikube...'
                bat 'minikube start'
            }
        }

        stage('Load the image to minikube') {
            steps {
                echo 'Load the image to minikube...'
                bat 'minikube image load pms-app:latest'
            }
        }
	stage('Deploy to k8') {
            steps {
                echo 'Deploy to k8...'
                bat 'kubectl apply -f k8s/'

		echo 'Checking Deployments...'
		bat 'kubectl get deployments'

		echo 'Checking for PODS...'
		bat 'kubectl get pods'

		echo 'Checking for Services'
		bat 'kubectl get svc'
            }
        }

	stage('Wait for the pods to be ready ') {
            steps {
                echo 'Wait for the pods to be ready...'
                bat 'kubectl wait --for=condition=Ready pods --all --timeout=180s'

            }
        }
	stage('Access the application') {
            steps {
                echo 'Access the application using URL...'
                bat 'minikube service pms-service --url'
            }
        }
    }
    
    post {
        always {
            echo 'This will always run after the pipeline finishes, regardless of status.'
        }
        success {
           
            echo 'Deployment successful!'
        }
        failure {
            echo 'Deployment failed, sending alert...'
        }
    }
}
